PR5: Data Preprocessing and Exploratory Data Analysis

Dataset: Titanic (train.csv)

import pandas as pd import numpy as np import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split from
sklearn.preprocessing import StandardScaler, LabelEncoder

———————————————————

1. Load dataset

———————————————————

df = pd.read_csv(“train.csv”)

print(“First Five Records”) print(df.head())

print(“Information”) df.info()

———————————————————

2. Remove duplicate rows

———————————————————

df.drop_duplicates(inplace=True)

print(“Rows:”, df.duplicated().sum())

———————————————————

3. Handle missing values

———————————————————

print(“Values Before Handling”) print(df.isnull().sum())

Fill missing Age values with the median

df[“Age”] = df[“Age”].fillna(df[“Age”].median())

Fill missing Embarked values with the most frequent value

df[“Embarked”] = df[“Embarked”].fillna(df[“Embarked”].mode()[0])

Cabin has a large number of missing values, so remove the column

df.drop(columns=[“Cabin”], inplace=True)

print(“Values After Handling”) print(df.isnull().sum())

———————————————————

4. Encode categorical variables

———————————————————

Convert Sex into numerical values

label_encoder = LabelEncoder() df[“Sex”] =
label_encoder.fit_transform(df[“Sex”])

Convert Embarked into dummy/indicator columns

df = pd.get_dummies(df, columns=[“Embarked”], drop_first=True,
dtype=int)

Remove text columns that are not used in this analysis

df.drop(columns=[“Name”, “Ticket”], inplace=True)

print(“After Encoding”) print(df.head())

———————————————————

5. Statistical summary and correlation

———————————————————

print(“Summary”) print(df.describe())

print(“Matrix”) print(df.corr(numeric_only=True))

———————————————————

6. Data visualization

———————————————————

Histogram of Age

plt.figure(figsize=(10, 5)) df[“Age”].hist(bins=20) plt.title(“Histogram
of Age”) plt.xlabel(“Age”) plt.ylabel(“Frequency”) plt.tight_layout()
plt.show()

Boxplot of Age

plt.figure(figsize=(6, 5)) sns.boxplot(y=df[“Age”]) plt.title(“Boxplot
of Age”) plt.tight_layout() plt.show()

Boxplot of Fare

plt.figure(figsize=(6, 5)) sns.boxplot(y=df[“Fare”]) plt.title(“Boxplot
of Fare”) plt.tight_layout() plt.show()

———————————————————

7. Separate features and target

———————————————————

X = df.drop(columns=[“Survived”]) y = df[“Survived”]

———————————————————

8. Split dataset into training and testing sets

———————————————————

X_train, X_test, y_train, y_test = train_test_split( X, y,
test_size=0.20, random_state=42 )

print(“Set Shape:”, X_train.shape) print(“Testing Set Shape :”,
X_test.shape) print(“Labels:”, y_train.shape) print(“Testing Labels :”,
y_test.shape)

———————————————————

9. Feature scaling

———————————————————

Fit the scaler only on training data to avoid data leakage

numerical_columns = [“Age”, “Fare”]

scaler = StandardScaler()

X_train = X_train.copy() X_test = X_test.copy()

X_train.loc[:, numerical_columns] = scaler.fit_transform(
X_train[numerical_columns] )

X_test.loc[:, numerical_columns] = scaler.transform(
X_test[numerical_columns] )

print(“Training Data”) print(X_train.head())

———————————————————

10. Final dataset information

———————————————————

print(“Training Dataset”) print(X_train.head())

print(“Testing Dataset”) print(X_test.head())

print(“Features Shape:”, X_train.shape) print(“Testing Features Shape
:”, X_test.shape)

print(“and EDA completed successfully.”)
